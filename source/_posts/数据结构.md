### 插入排序

#### 直接插入排序

```python
"""
前半段nums[0]~nums[i-1]是有序段；后半段nums[i]~nums[n]是“无序段”
时间复杂度:最好情况O(n);最坏情况O($n^2$);平均情况O(n^2)
稳定排序
"""
class Solution:
def insert(self, n, nums):
for i in range(n):
for j in range(i):
if nums[i] < nums[j]:
nums[i], nums[j] = nums[j], nums[i]
return nums
```

#### 折半插入排序

```python
"""
i为待排序元素位置
在0~i-1中进行判断
mid = (l + r + 1)//2
>: r = mid - 1
<: l = mid + 1
r+1~i-1往后挪一位，i位置元素插入r+1
减少了比较次数，没有减少移动次数
"""
class Solution:
def hinsert(self, n, nums):
if nums[1] < nums[0]: nums[1], nums[0] = nums[0], nums[1]
for i in range(2, n):
l, r = 0, i - 1
temp = nums[i]
while l <= r:
mid = (l + r + 1) // 2
if nums[mid] > temp: r = mid - 1
else: l = mid + 1
for j in range(i-1, r, -1): nums[j+1] = nums[j]
nums[r+1] = temp
return nums
```

#### 堆排序

```python
"""
根最小为小根堆；根最大为大根堆
如果需要在大一个数组中找到前100个最大的数，只需要维护一个包含100个元素的小根堆即可
"""
class Solution:
def heap(self, nums, heap_size, parent_index):
# temp保存需要下沉的父节点，用于最后赋值
temp = heap[parent_index]
child_index = 2*parent_index + 1
while child_index < heap_size:
# 如果有右孩子，且右孩子比左孩子大，则定位到右孩子
if child_index+1 < heap_size and nums[child_index+1] > nums[child_index]: child_index += 1
# 如果父节点的值不小于左右孩子节点的值，可直接跳出循环
if nums[parent_index] > nums[child_index]: break
nums[parent_index] = nums[child_index]
parent_index = child_index
child_index = 2*parent_index + 1
heap[child_index] = temp
def Sort(self, n, nums):
# 1. 无序列表构建成最大堆
for i in range(n-1//2, -1, -1):
self.heap(nums, n, i)
# 2. 循环删除堆顶元素，移到列表尾部，调节堆产生新的堆顶
for i in range(n-1, 0, -1):
self.heap(nums, i, 0)
```

for i in range(size): # 遍历这一层的所有节点
node = q.popleft() # 取出队首元素
if i < size-1: node.next = q[0] # 连接
# 拓展下一层的节点
if node.left: q.append(node.left)
if node.right: q.append(node.right)
# 返回根节点

while head:

#### leetcode39

题目描述：

```plain
给定一个无重复元素的数组candidates和一个目标数target，找出candidates中所有可以使数字和为target的组合。candidates中的数字可以无限制重复被选取。
说明：
所有数字（包括target）都是正整数。

```

```plain

```

代码：

```python
class Solution:
def combinationSum(self, candidates, target):
if not candidates:  #先解决空输入的情况
return []
candidates.sort()  #排序
n = len(candidates)
res = []
def backtrack(i, temp_sum, temp_list):
"""
i：遍历到candidates数组中第几个元素
temp_sum：目前遍历数组的和
temp_list：目前遍历的数组
"""
if temp_sum == target:
res.append(temp_list)
return
if temp_sum > target: return
for j in range(i, n):
backtrack(j, temp_sum+candidates[j], temp_list+[candidates[j]])
backtrack(0, 0, [])
return res
```

class Solution:
def Water(self, n, height):
l, r = [], []
l[0], r[0], ans = height[0], height[-1], 0
for i in range(1, n):
if height[i] >= height[i-1]: l.append(height[i])
else: l.append(l[-1])
for i in range(n-1, 0, -1):
if height[i] >= height[i+1]: r.append(height[i])
else: r.append(r[-1])
